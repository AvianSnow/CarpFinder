    /// <summary>Runs the world in half seconds, writing a line whenever a creature's state changes.</summary>
    private async Task Watch(int seconds, IReadOnlyList<EntityUid> mobs, Action<float>? script = null)
    {
        var start = TimeSpan.Zero;
        await Server.WaitPost(() => start = Timing.CurTime);
        var steps = seconds * 2;
        for (var i = 0; i < steps; i++)
        {
            await Pair.RunTicksSync(Timing.TickRate / 2);
            await Server.WaitPost(() =>
            {
                var t = (float) (Timing.CurTime - start).TotalSeconds;
                script?.Invoke(t);
                foreach (var mob in mobs)
                {
                    if (!SEntMan.EntityExists(mob) || !SEntMan.HasComponent<CFGOAPComponent>(mob))
                        continue;

                    if (i % 20 == 0)
                        _out.AppendLine(FormattableString.Invariant($"{t,5:F1} {mob} BODY {Body(mob)}"));

                    if (SEntMan.System<MobStateSystem>().IsDead(mob))
                        continue;

                    if (!_paths.TryGetValue(mob, out var path))
                        _paths[mob] = path = new Dictionary<Vector2i, int>();

                    path.TryAdd(TileOf(mob), (int) t);
                    var line = Line(mob);
                    var key = line[(line.IndexOf(' ') + 1)..];
                    key = key[(key.IndexOf(' ') + 1)..];
                    if (_lastLine.GetValueOrDefault(mob) == key && i % 8 != 0)
                        continue;

                    _lastLine[mob] = key;
                    _out.AppendLine(FormattableString.Invariant($"{t,5:F1} {mob} {line}"));
                }
            });
        }
    }

    private char Glyph(Vector2i tile, MapGridComponent grid, Dictionary<Vector2i, int>? path)
    {
        var glyph = Maps.TryGetTileRef(_grid, grid, tile, out var tileRef) && !tileRef.Tile.IsEmpty ? '.' : ' ';
        foreach (var e in Maps.GetAnchoredEntities(_grid, grid, tile))
        {
            var id = Name(e);
            if (id.StartsWith("Wall"))
                glyph = '#';
            else if (id == "Window" || id.Contains("WindowWooden"))
                glyph = 'W';
            else if (id == "Grille" && glyph != 'W')
                glyph = 'g';
            else if (id == "GrilleBroken")
                glyph = ',';
            else if (SEntMan.TryGetComponent(e, out DoorComponent? door))
                glyph = door.State is DoorState.Open or DoorState.Opening ? 'o' : 'D';
            else if (id.StartsWith("Table"))
                glyph = 't';
        }

        if (path != null && path.TryGetValue(tile, out var when) && glyph is '.' or 'o' or ',' or 'g')
            glyph = "0123456789abcdefghijklmnopqrstuvwxyz"[Math.Min(35, when / 5)];

        return glyph;
    }

    private void Report(string name, IEnumerable<EntityUid> mobs, IEnumerable<EntityUid> heroes)
    {
        var text = new StringBuilder();
        text.AppendLine($"===== {name} =====");
        text.Append(_out);
        var grid = SEntMan.GetComponent<MapGridComponent>(_grid);
        foreach (var hero in heroes)
        {
            if (!SEntMan.TryGetComponent(hero, out CFCombatantComponent? combat))
                continue;

            text.AppendLine($"hero {hero} lost {combat.MaxHp - combat.CurrentHp} hp, at {TileOf(hero)}");
        }

        foreach (var mob in mobs)
        {
            if (SEntMan.EntityExists(mob))
                text.AppendLine($"end {mob} {Body(mob)}");
        }

        var bridge = SEntMan.System<CFCombatBridgeSystem>();
        var shown = new HashSet<string>();
        var anchored = SEntMan.AllEntityQueryEnumerator<TransformComponent>();
        while (anchored.MoveNext(out var uid, out var xform))
        {
            var id = Name(uid);
            if (xform.Anchored && id is "Grille" or "Window" or "CFWoodenBirchDoor" && shown.Add(id)
                && bridge.TryEnsureCombatTarget(uid, out var combat, out var hardness, out var isObject))
                text.AppendLine($"object {id}: hp {combat.CurrentHp}/{combat.MaxHp} ac {combat.ArmorClass} hardness {hardness} isObject {isObject}");
        }

        SEntMan.System<CFGOAPDebugLogSystem>().WatchAll();
        foreach (var mob in mobs)
        {
            if (!SEntMan.EntityExists(mob))
                continue;

            if (SEntMan.TryGetComponent(mob, out CFGOAPDebugLogComponent? log))
            {
                text.AppendLine($"--- decision log {mob} {Name(mob)}");
                foreach (var entry in log.Entries)
                {
                    text.AppendLine("  " + entry);
                }
            }

            _paths.TryGetValue(mob, out var path);
            text.AppendLine($"--- path {mob} (digit = 5 s blocks), {path?.Count ?? 0} tiles");
            for (var y = 9; y >= -17; y--)
            {
                var row = new StringBuilder(FormattableString.Invariant($"{y,4} "));
                for (var x = -11; x <= 15; x++)
                {
                    row.Append(Glyph(new Vector2i(x, y), grid, path));
                }

                text.AppendLine(row.ToString());
            }
        }

        File.WriteAllText(Path.Combine(OutDir, $"trial-{name}.txt"), text.ToString());
    }

    /// <summary>Runs the real command, and reports what it changed on each tile.</summary>
    [Test]
    public async Task FixGridAtmosCompare()
    {
        await Load();
        await Pair.RunTicksSync(30);
        var before = new Dictionary<Vector2i, string>();
        var text = new StringBuilder();
        await Server.WaitPost(() =>
        {
            foreach (var (tile, reading) in AirReadings())
            {
                before[tile] = reading;
            }

            var kinds = before.Values.GroupBy(v => v).Select(g => $"{g.Count()} x {g.Key}");
            text.AppendLine("before: " + string.Join(" | ", kinds));
            Server.ResolveDependency<Robust.Shared.Console.IConsoleHost>().ExecuteCommand($"fixgridatmos {SEntMan.GetNetEntity(_grid)}");
        });
        await Pair.RunTicksSync(30);
        await Server.WaitPost(() =>
        {
            var after = AirReadings();
            var kinds = after.Values.GroupBy(v => v).Select(g => $"{g.Count()} x {g.Key}");
            text.AppendLine("after:  " + string.Join(" | ", kinds));
            var changed = after.Where(p => before.GetValueOrDefault(p.Key) != p.Value).ToList();
            text.AppendLine($"tiles changed: {changed.Count}");
            foreach (var (tile, reading) in changed.Take(12))
            {
                text.AppendLine($"  {tile}: {before.GetValueOrDefault(tile)} became {reading}");
            }

            var solid = 0;
            var solidNoAir = 0;
            var grid = SEntMan.GetComponent<MapGridComponent>(_grid);
            foreach (var (tile, reading) in after)
            {
                if (reading != "none")
                    continue;

                solidNoAir++;
                if (Maps.GetAnchoredEntities(_grid, grid, tile).Any(e => Name(e) is "WallReinforced" or "Window" or "Grille" or "CFWoodenBirchDoor"))
                    solid++;
            }

            text.AppendLine($"tiles with no air: {solidNoAir}, of which under a wall, window, grille or door: {solid}");
            File.WriteAllText(Path.Combine(OutDir, "trial-fixgridatmos.txt"), text.ToString());
            Server.ResolveDependency<IConfigurationManager>().SetCVar(CCVars.NPCEnabled, false);
        });
    }

    private Dictionary<Vector2i, string> AirReadings()
    {
        var atmos = SEntMan.System<AtmosphereSystem>();
        var grid = SEntMan.GetComponent<MapGridComponent>(_grid);
        var result = new Dictionary<Vector2i, string>();
        foreach (var tile in Maps.GetAllTiles(_grid, grid))
        {
            var air = atmos.GetTileMixture(_grid, null, tile.GridIndices);
            result[tile.GridIndices] = air == null
                ? "none"
                : FormattableString.Invariant($"{air.Pressure:F0}kPa {air.Temperature:F0}K");
        }

        return result;
    }
